// 1. Azure 설정 (이 값들을 환경 변수로 설정해야 함)
// firebase functions:config:set ms.client_id="YOUR_ID" ms.client_secret="YOUR_SECRET" ms.tenant_id="YOUR_TENANT"
const MS_CONFIG = {
  auth: {
    clientId: functions.config().ms.client_id,
    authority: `https://login.microsoftonline.com/${functions.config().ms.tenant_id}`,
    clientSecret: functions.config().ms.client_secret,
  },
};

const pca = new ConfidentialClientApplication(MS_CONFIG);
const SCOPES = ["OnlineMeetings.ReadWrite", "User.Read", "Offline_access"];

// 2. [함수 A] 로그인 URL 생성 (프론트엔드에서 호출)
exports.getMSAuthUrl = functions.https.onCall(async (data, context) => {
  if (!context.auth) throw new functions.https.HttpsError('unauthenticated', 'Login required');
  
  const authCodeUrlParameters = {
    scopes: SCOPES,
    redirectUri: data.redirectUrl, // 예: https://your-app.com/portal/settings
  };

  const response = await pca.getAuthCodeUrl(authCodeUrlParameters);
  return { url: response };
});
